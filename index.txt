        function updateS3() {
            let rate = parseInt(s3Rate.value);
            let bpc = parseInt(s3Bit.value);
            
            document.getElementById('s3-bit-val').innerText = bpc + " bits";
            document.getElementById('s3-levels').innerText = Math.pow(2, bpc).toLocaleString();
            
            let bytesPerSec = (rate * bpc) / 8;
            document.getElementById('s3-size-kb').innerText = (bytesPerSec / 1024).toFixed(1);
            document.getElementById('s3-size-bytes').innerText = bytesPerSec.toLocaleString() + " bytes/s";
            
            let nyqWarn = document.getElementById('s3-nyquist-warn');
            let nyqOk = document.getElementById('s3-nyquist-ok');
            let passNyq = rate > 880;
            if(passNyq) { nyqWarn.classList.add('hidden'); nyqOk.classList.remove('hidden'); }
            else { nyqWarn.classList.remove('hidden'); nyqOk.classList.add('hidden'); }
            
            document.getElementById('s3-play-err').classList.toggle('hidden', rate >= 8000);
            
            // Draw
            let cw = s3Canvas.parentElement.clientWidth;
            let ch = s3Canvas.parentElement.clientHeight;
            s3Canvas.width = cw; s3Canvas.height = ch;
            
            let freq = 440;
            let periods = 2;
            let totalTime = periods / freq; // ~0.0045s
            
            s3Ctx.clearRect(0,0,cw,ch);
            
            // Midline
            s3Ctx.strokeStyle = isDark ? "#27272a" : "#e2e8f0";
            s3Ctx.beginPath(); s3Ctx.moveTo(0, ch/2); s3Ctx.lineTo(cw, ch/2); s3Ctx.stroke();
            
            // Original thin wave
            s3Ctx.strokeStyle = isDark ? "rgba(255,255,255,0.2)" : "rgba(0,0,0,0.2)";
            s3Ctx.lineWidth = 1;
            s3Ctx.beginPath();
            for(let x=0; x<=cw; x++) {
                let t = (x/cw) * totalTime;
                let y = Math.sin(2 * Math.PI * freq * t);
                let py = ch/2 - (y * ch/2.2);
                if(x===0) s3Ctx.moveTo(x,py); else s3Ctx.lineTo(x,py);
            }
            s3Ctx.stroke();
            
            // Quantized points & steps
            let levels = Math.pow(2, bpc);
            let dt = 1/rate;
            let points = [];
            for(let t=0; t<=totalTime; t+=dt) {
                let origY = Math.sin(2 * Math.PI * freq * t);
                let qY = Math.floor((origY + 1) / 2 * (levels - 1));
                qY = (qY / (levels - 1)) * 2 - 1;
                let x = (t/totalTime) * cw;
                let py = ch/2 - (qY * ch/2.2);
                points.push({x, py});
            }
            document.getElementById('s3-points-info').innerText = `จุดสุ่ม: ${Math.round(rate/freq)} จุดต่อคาบ`;

            s3Ctx.strokeStyle = isDark ? "#a78bfa" : "#8b5cf6"; // Primary
            s3Ctx.lineWidth = 2;
            s3Ctx.beginPath();
            if(points.length > 0) {
                s3Ctx.moveTo(points[0].x, points[0].py);
                for(let i=1; i<points.length; i++) {
                    s3Ctx.lineTo(points[i].x, points[i-1].py); // hold
                    s3Ctx.lineTo(points[i].x, points[i].py);   // step
                }
                s3Ctx.lineTo(cw, points[points.length-1].py);
                s3Ctx.stroke();
                
                // Dots
                s3Ctx.fillStyle = isDark ? "#f87171" : "#ef4444";
                points.forEach(p => {
                    s3Ctx.beginPath(); s3Ctx.arc(p.x, p.py, 3, 0, Math.PI*2); s3Ctx.fill();
                });
            }

            // Mission
            let m1Pass = passNyq;
            let m2Pass = bpc >= 8;
            let m3Pass = bytesPerSec <= 20480;
            
            document.getElementById('s3-m1').className = m1Pass ? "text-success flex gap-2" : "text-error flex gap-2";
            document.getElementById('s3-m2').className = m2Pass ? "text-success flex gap-2" : "text-error flex gap-2";
            document.getElementById('s3-m3').className = m3Pass ? "text-success flex gap-2" : "text-error flex gap-2";
            
            let mbBox = document.getElementById('s3-mission');
            let msBox = document.getElementById('s3-status');
            if(m1Pass && m2Pass && m3Pass) {
                mbBox.classList.add('mission-success');
                msBox.innerText = "สำเร็จ! เสียงคมชัดและขนาดเหมาะสม";
                msBox.className = "text-sm font-bold text-success";
                if(!progress.s3) { progress.s3 = true; saveProgress(); }
            } else {
                mbBox.classList.remove('mission-success');
                msBox.innerText = "กำลังทำภารกิจ...";
                msBox.className = "text-sm font-bold text-error";
            }
        }
        
        s3Btn.addEventListener('click', () => {
            let rate = parseInt(s3Rate.value);
            if(rate < 8000) return; // Browser constraint
            
            if(!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            if(audioCtx.state === 'suspended') audioCtx.resume();
            
            let bpc = parseInt(s3Bit.value);
            let buffer = audioCtx.createBuffer(1, rate, rate); // 1 sec
            let data = buffer.getChannelData(0);
            let levels = Math.pow(2, bpc);
            
            for(let i=0; i<rate; i++) {
                let t = i / rate;
                let val = Math.sin(2 * Math.PI * 440 * t);
                let qVal = Math.floor((val + 1) / 2 * (levels - 1));
                val = (qVal / (levels - 1)) * 2 - 1;
                // apply simple envelope to avoid clicks
                let env = 1.0;
                if(i < 500) env = i/500;
                else if(i > rate-500) env = (rate-i)/500;
                data[i] = val * 0.15 * env;
            }
            
            let source = audioCtx.createBufferSource();
            source.buffer = buffer;
            source.connect(audioCtx.destination);
            source.start();
        });

        function updateS4() {

            let compSlider = parseFloat(s4Comp.value); // 0-100
            
            // Log scale 1 to 300
            let min = 1, max = 300;
            let ratio = Math.exp(Math.log(min) + (compSlider / 100) * (Math.log(max) - Math.log(min)));
            
            document.getElementById('s4-fps-val').innerText = fps + " fps";
            document.getElementById('s4-comp-val').innerText = ratio.toFixed(1) + " : 1";
            document.getElementById('s4-canvas-fps').innerText = fps;
            
            let warn = document.getElementById('s4-comp-warn');
            if(ratio < 10) { warn.innerText = "ขนาดใหญ่มาก! ไม่เหมาะกับเว็บ"; warn.className = "text-xs mt-1 text-right text-error"; }
            else if(ratio < 50) { warn.innerText = "คุณภาพสูง ขนาดปานกลาง"; warn.className = "text-xs mt-1 text-right text-primary"; }
            else { warn.innerText = "บีบอัดสูง อาจเสียรายละเอียด"; warn.className = "text-xs mt-1 text-right text-muted"; }
            
            // Raw frame: 1280x720x3 = 2,764,800 bytes
            let rawBytes = 2764800;
            let rawBitsPerSec = rawBytes * fps * 8;
            let rawMbit = rawBitsPerSec / 1000000;
            
            let compBitsPerSec = rawBitsPerSec / ratio;
            let compMbit = compBitsPerSec / 1000000;
            
            let sizePerMinMB = (compBitsPerSec * 60) / 8 / 1000000;
            
            document.getElementById('s4-raw-bitrate').innerText = rawMbit.toFixed(2) + " Mbit/s";
            document.getElementById('s4-bitrate-mb').innerText = compMbit.toFixed(2);
            document.getElementById('s4-size-min').innerText = sizePerMinMB.toFixed(2) + " MB";
            
            // Mission
            let m1Pass = fps >= 24;
            let m2Pass = compMbit <= 5.0;
            
            document.getElementById('s4-m1').className = m1Pass ? "text-success flex gap-2" : "text-error flex gap-2";
            document.getElementById('s4-m2').className = m2Pass ? "text-success flex gap-2" : "text-error flex gap-2";
            
            let mbBox = document.getElementById('s4-mission');
            let msBox = document.getElementById('s4-status');
            if(m1Pass && m2Pass) {
                mbBox.classList.add('mission-success');
                msBox.innerText = "สำเร็จ! ได้วิดีโอที่สตรีมได้ไหลลื่น";
                msBox.className = "text-sm font-bold text-success";

                msBox.innerText = "กำลังทำภารกิจ...";
                msBox.className = "text-sm font-bold text-error";
            }
        }
        
        function drawS4Frame(yPos) {
            let cw = s4Canvas.parentElement.clientWidth;
            let ch = cw * (9/16);
            s4Canvas.width = cw; s4Canvas.height = ch;
            
            // BG
            s4Ctx.fillStyle = isDark ? "#18181b" : "#f8fafc";
            s4Ctx.fillRect(0,0,cw,ch);
            
            // Grid
            s4Ctx.strokeStyle = isDark ? "#27272a" : "#e2e8f0";
            s4Ctx.lineWidth = 1;
            for(let i=0; i<cw; i+=40) { s4Ctx.beginPath(); s4Ctx.moveTo(i,0); s4Ctx.lineTo(i,ch); s4Ctx.stroke(); }
            
            // Ball (x is center, y bounces)
            let r = ch * 0.15;
            let drawY = r + yPos * (ch - r*2); 
            s4Ctx.fillStyle = isDark ? "#a78bfa" : "#8b5cf6";
            s4Ctx.beginPath(); s4Ctx.arc(cw/2, drawY, r, 0, Math.PI*2); s4Ctx.fill();
        }

        function animS4(time) {
            if(!s4StartTime) s4StartTime = time;
            
            if(prefMotion) {
                drawS4Frame(0.5); // static
                return;
            }
            
            let elapsed = (time - s4StartTime) / 1000;
            let fps = parseInt(s4Fps.value);
            let currentFrame = Math.floor(elapsed * fps);
            
            if(currentFrame > s4LastDrawnFrame) {
                // Physics: 1 sec down, 1 sec up
                let t = elapsed % 2.0;
                let yPos = t < 1.0 ? t : (2.0 - t); // 0 to 1 back to 0
                
                // Add easing (bounce) - simple quadratic
                if(t < 1.0) yPos = Math.pow(t, 2);
                else yPos = Math.pow(2.0-t, 2);
                
                drawS4Frame(yPos);
                s4LastDrawnFrame = currentFrame;
                document.getElementById('s4-canvas-f').innerText = currentFrame;
                document.getElementById('s4-canvas-t').innerText = elapsed.toFixed(2);
            }
            
            s4AnimId = requestAnimationFrame(animS4);
        }

        function renderQuiz() {

                div.appendChild(qText);
                
                let grid = document.createElement('div');
                grid.className = "grid grid-cols-1 sm:grid-cols-2 gap-3";
                
                item.opts.forEach((opt, oIdx) => {

                    btn.className = "p-3 border border-border rounded-lg text-left transition-colors hover:bg-bg";
                    btn.innerText = opt;
                    
                    if(quizAnswers[qIdx] === oIdx) {
                        if(oIdx === item.ans) {
                            btn.className = "p-3 border-2 border-success bg-success/10 text-success rounded-lg text-left font-medium";
                        } else {
                            btn.className = "p-3 border-2 border-error bg-error/10 text-error rounded-lg text-left font-medium";
                        }
                    } else if(quizAnswers[qIdx] !== -1 && oIdx === item.ans) {
                        btn.className = "p-3 border border-success text-success rounded-lg text-left";
                    }
                    
                    btn.onclick = () => {
                        if(quizAnswers[qIdx] !== -1) return; // Answered
                        quizAnswers[qIdx] = oIdx;
                        renderQuiz();
                        checkQuizCompletion();
                    };
                    grid.appendChild(btn);
                });

                if(quizAnswers[qIdx] !== -1) {
                    let exp = document.createElement('div');
                    exp.className = `mt-4 p-3 text-sm rounded ${quizAnswers[qIdx] === item.ans ? 'bg-success/10 text-success' : 'bg-error/10 text-error'}`;
                    exp.innerHTML = `<b>คำอธิบาย:</b> ${item.exp}`;
                    div.appendChild(exp);
                }
                
                c.appendChild(div);
            });
        }
